// ===== Persistent state: browser and Electron | src/App.tsx:553-615 =====
function usePersistentState<T>(key: string, initialValue: T) {
  const localStorageHadValue = useRef(typeof window !== 'undefined' && window.localStorage.getItem(key) !== null);
  const [state, setState] = useState<T>(() => {
    if (typeof window === 'undefined') return initialValue;
    const saved = window.localStorage.getItem(key);
    if (!saved) return initialValue;
    try {
      return JSON.parse(saved) as T;
    } catch {
      return initialValue;
    }
  });
  const initialState = useRef(state);
  const desktopLoaded = useRef(!window.bayyinahDesktop);
  const [loaded, setLoaded] = useState(!window.bayyinahDesktop);
  const [loadError, setLoadError] = useState(false);

  useEffect(() => {
    const desktop = window.bayyinahDesktop;
    if (!desktop) return;
    let cancelled = false;
    desktopLoaded.current = false;
    setLoaded(false);
    setLoadError(false);

    void desktop.storage.get<T>(key).then(async (stored) => {
      if (cancelled) return;
      if (stored === null) {
        await desktop.storage.set(key, initialState.current);
      } else {
        setState(stored);
      }
      desktopLoaded.current = true;
      setLoaded(true);
    }).catch(() => {
      if (cancelled) return;
      if (key === 'bayyinah-users' && localStorageHadValue.current) {
        void desktop.storage.set(key, initialState.current);
        desktopLoaded.current = true;
        setLoaded(true);
        return;
      }
      desktopLoaded.current = true;
      setLoadError(true);
      setLoaded(true);
    });

    return () => {
      cancelled = true;
    };
  }, [key]);

  useEffect(() => {
    const desktop = window.bayyinahDesktop;
    if (desktop) {
      if (desktopLoaded.current) void desktop.storage.set(key, state);
      return;
    }
    window.localStorage.setItem(key, JSON.stringify(state));
  }, [key, state]);

  return [state, setState, loaded, loadError] as const;
}

function getDaysUntil(date: string) {
  if (!date) return Number.POSITIVE_INFINITY;
  const target = new Date(date);
  const today = new Date();
  target.setHours(0, 0, 0, 0);
  today.setHours(0, 0, 0, 0);
  return Math.round((target.getTime() - today.getTime()) / 86400000);
}

function toIsoDate(date: Date) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

function todayIsoDate() {
  return toIsoDate(new Date());
}

// ===== Deferred SQLite disk persistence | electron/main.cjs:202-212 =====
function persistDatabase() {
  if (!database || !databasePath) return;
  if (persistTimer) clearTimeout(persistTimer);
  persistTimer = undefined;
  fs.writeFileSync(databasePath, Buffer.from(database.export()));
}

function schedulePersist() {
  if (persistTimer) clearTimeout(persistTimer);
  persistTimer = setTimeout(persistDatabase, 150);
}

// ===== Payload encryption and decoding | electron/main.cjs:262-273 =====
function encodePayload(value) {
  const json = JSON.stringify(value);
  if (!safeStorage.isEncryptionAvailable()) return json;
  return `enc:v1:${safeStorage.encryptString(json).toString('base64')}`;
}

function decodePayload(value) {
  const payload = String(value ?? 'null');
  if (!payload.startsWith('enc:v1:')) return JSON.parse(payload);
  const encrypted = Buffer.from(payload.slice(7), 'base64');
  return JSON.parse(safeStorage.decryptString(encrypted));
}

// ===== Storage read write export IPC | electron/main.cjs:351-381 =====
function registerIpcHandlers() {
  ipcMain.handle('storage:get', (_event, key) => {
    const row = readScalar(
      `SELECT payload FROM records WHERE entity_type = 'app-state' AND record_id = ?`,
      [key],
    );
    return row ? decodePayload(row.payload) : null;
  });

  ipcMain.handle('storage:set', (_event, key, value) => {
    database.run(
      `INSERT INTO records(entity_type, record_id, payload, created_at, updated_at)
       VALUES('app-state', ?, ?, CURRENT_TIMESTAMP, CURRENT_TIMESTAMP)
       ON CONFLICT(entity_type, record_id)
       DO UPDATE SET payload = excluded.payload, updated_at = CURRENT_TIMESTAMP`,
      [key, encodePayload(value)],
    );
    schedulePersist();
    return true;
  });

  ipcMain.handle('storage:export', () => {
    const statement = database.prepare(`SELECT record_id, payload FROM records WHERE entity_type = 'app-state'`);
    const output = {};
    while (statement.step()) {
      const row = statement.getAsObject();
      output[row.record_id] = decodePayload(row.payload);
    }
    statement.free();
    return output;
  });
